def mult(a, b):
    return (a * .6) + (b * .4)

ex2 = int(input('Score exam 2: '))

result = mult(ex1,ex2)

print(f'Your first exam score is {ex1} and your second exam score is {ex2} . Your final overall score is  {result}')
