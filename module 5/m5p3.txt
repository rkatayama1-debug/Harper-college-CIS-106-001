def determineShippingCharge(cost):
    if (cost <=50.00):
        return (25.00)
    else:
        return (0.00)

nobook = int(input('number of books:'))
cpbook = float(input('cost per book:'))
totalcost = (cpbook) * (nobook)
shipcost = determineShippingCharge(totalcost)

print(f'Your purchase before shipping will total to : ${totalcost:.2f}\nYour shipping charge will be: ${shipcost:.2f}')
