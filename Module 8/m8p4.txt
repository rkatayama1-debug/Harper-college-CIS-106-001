totalExtendedPrice = 0
count = 0

file = open('orders.txt', 'r')

while True:
    item = file.readline().strip()

    if item == '':
        break

    quantity = int(file.readline())
    price = float(file.readline())

    extendedPrice = quantity * price

    priceString = "$" + "{:,.2f}".format(price)
    extendedPriceString = "$" + "{:,.2f}".format(extendedPrice)

    print(f'{item:<12}{quantity:>8}{priceString:>12}{extendedPriceString:>18}')

    totalExtendedPrice = totalExtendedPrice + extendedPrice
    count = count + 1

file.close()

print(f'\nTotal extended price: ${totalExtendedPrice:,.2f}')
print(f'Number of orders: {count}')

if count > 0:
    averageOrder = totalExtendedPrice / count
    print(f'Average order: ${averageOrder:,.2f}')
else:
    print('Average order: $0.00')
