
def calculate_rate(principle, years):
    if principle > 100000 and years == 5:
        rate = 0.06
    elif principle >= 50000 and principle <= 100000 and years == 10:
        rate = 0.05
    elif principle >= 50000 and principle <= 100000 and years == 5:
        rate = 0.04
    else:
        rate = 0.02

    return rate

principle = float(input('Enter principle amount: $'))
years = int(input('Enter years to maturity: '))

rate = calculate_rate(principle, years)
interest = principle * rate

print(f'Principle:           ${principle:>7,.2f}')
print(f'Interest rate:   {rate:>10.0%}')
print(f'First year interest: ${interest:>7,.2f}')
